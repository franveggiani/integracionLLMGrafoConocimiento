// ==========================
// Plan de Estudios – UTN-FRM
// Grafo base: Cursos + Áreas + Competencias + Semestres
// ==========================

// --- Restricciones de unicidad (idempotentes) ---
CREATE CONSTRAINT IF NOT EXISTS FOR (c:Course)   REQUIRE c.code IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (a:Area)     REQUIRE a.name IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (s:Skill)    REQUIRE s.name IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (sm:Semester) REQUIRE sm.num IS UNIQUE;
CREATE CONSTRAINT IF NOT EXISTS FOR (k:Category) REQUIRE k.name IS UNIQUE;

// --- Categorías para usar con :IS_A ---
MERGE (catUC:Category {name: "Unidad Curricular"});
MERGE (catSkill:Category {name: "Competencia"});

// --- Áreas (dominios) ---
UNWIND [
  "Programación",
  "Bases de Datos",
  "Arquitectura y SO",
  "Inteligencia Artificial"
] AS areaName
MERGE (:Area {name: areaName});

// --- Semestres (ej. 1 a 4) ---
UNWIND [1,2,3,4] AS semNum
MERGE (:Semester {num: semNum});

// --- Competencias (skills) ---
UNWIND [
  "Estructuras de Datos",
  "Algoritmos",
  "Programación Orientada a Objetos",
  "Patrones de Diseño",
  "SQL",
  "Modelado ER",
  "Concurrencia y Procesos",
  "Linux y CLI",
  "Aprendizaje Supervisado",
  "Evaluación de Modelos"
] AS skillName
MERGE (sk:Skill {name: skillName})
MERGE (sk)-[:IS_A]->(catSkill);

// --- Cursos (con metadatos mínimos) ---
UNWIND [
  {code:"IS-101", name:"Algoritmos y Estructuras de Datos", area:"Programación",          semester:1},
  {code:"IS-201", name:"Programación Avanzada",              area:"Programación",          semester:2},
  {code:"IS-202", name:"Ingeniería de Software I",           area:"Programación",          semester:3},
  {code:"IS-301", name:"Bases de Datos",                     area:"Bases de Datos",        semester:2},
  {code:"IS-302", name:"Arquitectura de Computadoras",       area:"Arquitectura y SO",     semester:2},
  {code:"IS-303", name:"Sistemas Operativos",                area:"Arquitectura y SO",     semester:3},
  {code:"IS-401", name:"Inteligencia Artificial",            area:"Inteligencia Artificial",semester:4},
  {code:"IS-402", name:"Minería de Datos",                   area:"Inteligencia Artificial",semester:4}
] AS c
MERGE (course:Course {code: c.code})
  ON CREATE SET course.name = c.name
  ON MATCH  SET course.name = c.name
WITH course, c
MATCH (a:Area {name: c.area})
MATCH (sm:Semester {num: c.semester})
MATCH (catUC:Category {name: "Unidad Curricular"})
MERGE (course)-[:BELONGS_TO]->(a)
MERGE (course)-[:TAUGHT_IN]->(sm)
MERGE (course)-[:IS_A]->(catUC);

// --- Mapeo Curso -> Competencias cubiertas ---
UNWIND [
  {code:"IS-101", skills:["Algoritmos","Estructuras de Datos"]},
  {code:"IS-201", skills:["Programación Orientada a Objetos","Patrones de Diseño"]},
  {code:"IS-202", skills:["Programación Orientada a Objetos","Patrones de Diseño"]},
  {code:"IS-301", skills:["SQL","Modelado ER"]},
  {code:"IS-302", skills:["Concurrencia y Procesos","Linux y CLI"]},
  {code:"IS-303", skills:["Concurrencia y Procesos","Linux y CLI"]},
  {code:"IS-401", skills:["Algoritmos","Aprendizaje Supervisado","Evaluación de Modelos"]},
  {code:"IS-402", skills:["Aprendizaje Supervisado","Evaluación de Modelos","SQL"]}
] AS cs
MATCH (c:Course {code: cs.code})
UNWIND cs.skills AS skillName
MATCH (s:Skill {name: skillName})
MERGE (c)-[:COVERS]->(s);

// --- Correlación de prerrequisitos (PREREQ) ---
UNWIND [
  ["IS-101","IS-201"],   // Algoritmos y ED -> Programación Avanzada
  ["IS-201","IS-202"],   // Prog Avanzada -> Ingeniería de Software I
  ["IS-101","IS-301"],   // Algoritmos y ED -> Bases de Datos
  ["IS-302","IS-303"],   // Arquitectura -> Sistemas Operativos
  ["IS-101","IS-401"],   // Algoritmos y ED -> IA
  ["IS-301","IS-402"],   // Bases de Datos -> Minería de Datos
  ["IS-401","IS-402"]    // IA -> Minería de Datos
] AS pair
WITH pair[0] AS src, pair[1] AS dst
MATCH (c1:Course {code: src})
MATCH (c2:Course {code: dst})
MERGE (c1)-[:PREREQ]->(c2);

// (Opcional) Vista rápida: contar nodos y relaciones creadas/consolidadas
MATCH (n) RETURN labels(n) AS labels, count(*) AS total
ORDER BY total DESC;
